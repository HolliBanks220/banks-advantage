      if(ack && resolvedAt) resolutionMs.push(resolvedAt-ack);
    });

    return { awaiting:awaiting, awaitingOverdue:awaitingOverdue, inProgress:inProgress, inProgressOverdue:inProgressOverdue, awaitingConfirm:awaitingConfirm, verified:verified, reopenedTotal:reopenedTotal, avgFirstResponse:avg(firstResponseMs), avgResolution:avg(resolutionMs) };
  }

    dbWrite(db.from(TABLE).insert(row), "Couldn't submit that ticket");

  function findT(id){ for(var i=0;i<state.tickets.length;i++){ if(state.tickets[i].id===id) return state.tickets[i]; } return null; }

  function patchTicket(id, patch){ return dbWrite(db.from(TABLE).update(patch).eq("id", id), "Couldn't save that update"); }

  function actionAck(id, fields){
    var t = findT(id); if(!t) return;
    var updates = t.updates.concat([{ type:"ack", by: identity, at: nowIso(), etaDate: fields.etaDate||null, text: fields.text }]);
    setFormMode(id, null);
    patchTicket(id, { status:"acknowledged", updates: updates });
  }
  function actionResolve(id, fields){
    var t = findT(id); if(!t) return;
    var updates = t.updates.concat([{ type:"resolve", by: identity, at: nowIso(), text: fields.text }]);
    setFormMode(id, null);
    patchTicket(id, { status:"resolved", updates: updates });
  }
  function actionVerify(id, worked, fields){
    var t = findT(id); if(!t) return;
    var updates, patch;
    if(worked){
      updates = t.updates.concat([{ type:"verify", by: identity, at: nowIso(), text: (fields&&fields.text) || "Confirmed — this is working now." }]);
      patch = { status:"verified", updates: updates };
    } else {
      updates = t.updates.concat([{ type:"reopen", by: identity, at: nowIso(), text: fields.text }]);
      patch = { status:"reopened", updates: updates, reopen_count: (t.reopen_count||0)+1 };
    }
    setFormMode(id, null);
    patchTicket(id, patch);
  }
  function actionComment(id, fields){
    var t = findT(id); if(!t) return;
    var updates = t.updates.concat([{ type:"comment", by: identity, at: nowIso(), text: fields.text }]);
    setFormMode(id, null);
    patchTicket(id, { updates: updates });
  }
  function actionDelete(id){
    ui.confirmDeleteId = null;
    if(ui.expandedId===id) ui.expandedId=null;
    dbWrite(db.from(TABLE).delete().eq("id", id), "Couldn't delete that ticket");
  }

    dbWrite(db.from(TABLE).delete().in("id", ["WO-0001","WO-0002","WO-0003"]), "Couldn't clear examples");
  }

    window.location.href = mailto;
  }

  function filterTickets(){
    return state.tickets.filter(function(t){
      if(ui.filter==="all") return true;
      if(ui.filter==="active") return t.status!=="verified";
      return t.status===ui.filter;
    }).sort(function(a,b){
      var oa=sla(a).overdue?0:1, ob=sla(b).overdue?0:1;
      if(oa!==ob) return oa-ob;
      return new Date(b.created_at) - new Date(a.created_at);
    });
  }

  function renderList(){
    if(loading) return "<div class=\"loading\">Loading tickets…</div>";
    var list = filterTickets();
    if(list.length===0) return "<div class=\"empty\"><h3>Nothing here</h3><p>No tickets match this filter.</p></div>";
    return "<div class=\"list\">" + list.map(renderRow).join("") + "</div>";
  }

  function wireIdentityForm(){
    var f = document.getElementById("identity-form");
    if(!f) return;
    f.addEventListener("submit", function(e){
      e.preventDefault();
      var name = document.getElementById("id-name").value.trim();
      var role = document.getElementById("id-role").value;
      if(!name) return;
      saveIdentity({ name:name, role:role });
      render();
    });
  }

  function wireEvents(){
    var root = document.getElementById("app");
    root.querySelectorAll("[data-act]").forEach(function(node){
      node.addEventListener("click", function(e){
        var act = node.getAttribute("data-act");
        var id = node.getAttribute("data-id");
        if(act==="change-identity"){ identity=null; render(); return; }
        if(act==="filter"){ ui.filter=node.getAttribute("data-val"); render(); return; }
        if(act==="open-new"){ ui.newTicketOpen=true; render(); return; }
        if(act==="close-new"){ ui.newTicketOpen=false; render(); return; }
        if(act==="toggle"){ ui.expandedId = (ui.expandedId===id)?null:id; ui.formMode={}; render(); return; }
        if(act==="form"){ setFormMode(id, node.getAttribute("data-mode")); return; }
        if(act==="cancel-form"){ setFormMode(id, null); return; }
        if(act==="verify-yes"){ actionVerify(id, true, {}); return; }
        if(act==="verify-no"){ setFormMode(id, "reopen-note"); return; }
        if(act==="notify"){ var t=findT(id); if(t) notifyMailto(t, node.getAttribute("data-to")); return; }
        if(act==="delete"){ ui.confirmDeleteId=id; render(); return; }
        if(act==="cancel-delete"){ ui.confirmDeleteId=null; render(); return; }
        if(act==="confirm-delete"){ actionDelete(id); return; }
        if(act==="clear-examples"){ e.preventDefault(); actionClearExamples(); return; }
      });
    });
    var ntForm = document.getElementById("new-ticket-form");
    if(ntForm){
      ntForm.addEventListener("submit", function(e){
        e.preventDefault();
        var fields = { title: document.getElementById("nt-title").value.trim(), description: document.getElementById("nt-desc").value.trim(), category: document.getElementById("nt-cat").value, priority: document.getElementById("nt-pri").value };
        if(!fields.title || !fields.description) return;
        actionNewTicket(fields);
      });
    }
    root.querySelectorAll("form[data-form]").forEach(function(f){
      f.addEventListener("submit", function(e){
        e.preventDefault();
        var kind = f.getAttribute("data-form");
        var id = f.getAttribute("data-id");
        var textEl = document.getElementById("f-text-"+id);
        var text = textEl ? textEl.value.trim() : "";
        if(!text) return;
        if(kind==="ack"){ var etaEl=document.getElementById("f-eta-"+id); actionAck(id, { text:text, etaDate: etaEl?etaEl.value:"" }); }
        else if(kind==="resolve"){ actionResolve(id, { text:text }); }
        else if(kind==="reopen"){ actionVerify(id, false, { text:text }); }
        else if(kind==="comment"){ actionComment(id, { text:text }); }
      });
    });
  }
